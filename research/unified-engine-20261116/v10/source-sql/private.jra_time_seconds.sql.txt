CREATE OR REPLACE FUNCTION private.jra_time_seconds(p_time text)
 RETURNS numeric
 LANGUAGE sql
 IMMUTABLE
 SET search_path TO 'pg_catalog', 'public', 'private'
AS $function$
  select case
    when coalesce(p_time,'') ~ '^[0-9]+:[0-9]{2}\.[0-9]$' then
      split_part(p_time,':',1)::numeric * 60
      + split_part(split_part(p_time,':',2),'.',1)::numeric
      + ('0.' || split_part(p_time,'.',2))::numeric
    else null
  end;
$function$
