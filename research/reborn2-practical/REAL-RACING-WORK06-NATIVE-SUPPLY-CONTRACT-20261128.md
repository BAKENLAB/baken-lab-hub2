WITH RECURSIVE nodes(id,path,v,prefix) AS (
 SELECT id,ARRAY[]::int[],evidence,''::text FROM private.lab_preclaim_evidence_snapshots_v1
 UNION ALL SELECT n.id,n.path||ARRAY[2,x.ord::int],x.val,(CASE WHEN x.ord>1 THEN ',' ELSE '' END)||x.keyprefix
 FROM nodes n CROSS JOIN LATERAL (
 SELECT e.value AS val,row_number() OVER(ORDER BY e.key COLLATE "C") AS ord,to_jsonb(e.key)::text||':' AS keyprefix FROM jsonb_each(CASE WHEN jsonb_typeof(n.v)='object' THEN n.v ELSE '{}'::jsonb END) e
 UNION ALL SELECT a.value,a.ord,'' FROM jsonb_array_elements(CASE WHEN jsonb_typeof(n.v)='array' THEN n.v ELSE '[]'::jsonb END) WITH ORDINALITY a(value,ord)
 ) x
), tokens AS (
 SELECT id,path||0 AS path,prefix||CASE jsonb_typeof(v) WHEN 'object' THEN '{' WHEN 'array' THEN '[' WHEN 'number' THEN trim_scale((v#>>'{}')::numeric)::text ELSE v::text END AS token FROM nodes
 UNION ALL SELECT id,path||3,CASE jsonb_typeof(v) WHEN 'object' THEN '}' ELSE ']' END FROM nodes WHERE jsonb_typeof(v) IN ('object','array')
), hashed AS (SELECT id,encode(extensions.digest(string_agg(token,'' ORDER BY path),'sha256'),'hex') AS h FROM tokens GROUP BY id)
SELECT count(*) snapshots,count(*) FILTER(WHERE h=s.payload_digest) digest_match FROM hashed JOIN private.lab_preclaim_evidence_snapshots_v1 s USING(id);
