/* Official current card plus independently published JRA cancellation number; no guessed numbering. */
function parseCancellationNotice(raw,{date,track,raceNo,horseName}){
 const norm=s=>String(s??"").normalize("NFKC").replace(/\s+/g," ").trim();
 const esc=s=>s.replace(/[.*+?^${}()|[\]\\]/g,"\\$&");
 const s=norm(raw), [year,month,day]=date.split("-").map(Number);
 const head=new RegExp("第\\s*\\d+\\s*回\\s*"+esc(track)+"\\s*第\\s*\\d+\\s*日\\s*\\(\\s*"+month+"月"+day+"日\\s*\\([^)]*\\)\\s*\\)","g");
 const headings=[...s.matchAll(head)];
 if(headings.length!==1)throw Error("JRA_CANCEL_NOTICE_DATE_TRACK_UNVERIFIED");
 let section=s.slice(headings[0].index+headings[0][0].length);
 section=section.split(/第\s*\d+\s*回\s*(?:札幌|函館|福島|新潟|東京|中山|中京|京都|阪神|小倉)\s*第\s*\d+\s*日|今日の出来事|明日の取消・変更等/)[0];
 const marker=section.indexOf("出走取消");if(marker<0)throw Error("JRA_CANCEL_NOTICE_STATUS_UNVERIFIED");
 section=section.slice(marker+"出走取消".length).split(/騎手変更|発走時刻変更|競走除外/)[0];
 const entries=[...section.matchAll(new RegExp("(?:^|\\s)"+raceNo+"\\s*R\\s*(\\d{1,2})\\s*番\\s*"+esc(norm(horseName))+"\\s*\\(","g"))];
 if(entries.length!==1)throw Error("JRA_CANCEL_NOTICE_NUMBER_UNVERIFIED");
 const no=Number(entries[0][1]);if(no<1||no>24)throw Error("JRA_CANCEL_NOTICE_NUMBER_INVALID");
 return no;
}
export async function fetchCancellationNumber(load,args){
 if(args.date!=="2026-11-15"||args.track!=="京都"||args.raceNo!==10)throw Error("JRA_CANCEL_NOTICE_SOURCE_NOT_CONFIGURED");
 const url="https://www.jra.go.jp/news/202611/111404.html";
 const resp=await fetch(url,{headers:{"User-Agent":"Mozilla/5.0","Accept-Language":"ja-JP","Cache-Control":"no-cache"},redirect:"follow"});
 if(!resp.ok||!["www.jra.go.jp","jra.jp"].includes(new URL(resp.url).hostname))throw Error("JRA_CANCEL_NOTICE_FETCH_FAILED");
 const bytes=new Uint8Array(await resp.arrayBuffer());if(bytes.length<300||bytes.length>1500000)throw Error("JRA_CANCEL_NOTICE_CONTENT_INVALID");
 let result=null;
 for(const encoding of ["utf-8","shift_jis"]){
   const $=load(new TextDecoder(encoding).decode(bytes));
   $("body").find("br").replaceWith(" ");
   $("body").find("h1,h2,h3,h4,h5,p,li,dt,dd").each((_,el)=>{$(el).prepend(" ").append(" ")});
   try{result=parseCancellationNotice($("body").text(),args);break}catch{}
 }
 if(result===null)throw Error("JRA_CANCEL_NOTICE_NUMBER_UNVERIFIED");
 return result;
}
